3. Backend API & PostGIS

Node.js (Express) + Redis Cache + Proxy Carto Maps/SQL API bảo mật khóa API phía server + PostgreSQL/PostGIS quản lý điểm chụp.

server.js — Node.js / Express + Carto API Key + PostGIS + Redis Proxy 
/**
 * SkyRadar VN - Production Backend API Server (Node.js + Express + Carto GIS + Redis + PostGIS)
 * Biến môi trường (.env):
 * CARTO_API_KEY=cb1_501v_1_e33dc217369ec4a4d4697d9d
 * REDIS_URL=redis://127.0.0.1:6379
 * DATABASE_URL=postgres://postgres:password@localhost:5432/skyradar
 */
const express = require('express');
const cors = require('cors');
const Redis = require('ioredis');
const { Pool } = require('pg');

const app = express();
app.use(cors());
app.use(express.json());

const CARTO_API_KEY = process.env.CARTO_API_KEY || 'cb1_501v_1_e33dc217369ec4a4d4697d9d';
const redis = new Redis(process.env.REDIS_URL || 'redis://127.0.0.1:6379');
const pgPool = new Pool({ connectionString: process.env.DATABASE_URL });

// 1. GET /api/v1/carto/config - Trả về cấu hình Carto Basemap & trạng thái xác thực
app.get('/api/v1/carto/config', (req, res) => {
  res.json({
    provider: 'CARTO',
    authenticated: Boolean(CARTO_API_KEY),
    keyPreview: `${CARTO_API_KEY.slice(0, 10)}...${CARTO_API_KEY.slice(-4)}`,
    tileTemplates: {
      dark_all: `https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png?api_key=${CARTO_API_KEY}`,
      dark_nolabels: `https://{s}.basemaps.cartocdn.com/dark_nolabels/{z}/{x}/{y}{r}.png?api_key=${CARTO_API_KEY}`,
      voyager: `https://{s}.basemaps.cartocdn.com/rastertiles/voyager/{z}/{x}/{y}{r}.png?api_key=${CARTO_API_KEY}`
    }
  });
});

// 2. GET /api/v1/forecast?lat=21.2981&lon=104.4652 - Proxy & Cache Open-Meteo
app.get('/api/v1/forecast', async (req, res) => {
  try {
    const lat = parseFloat(req.query.lat).toFixed(3);
    const lon = parseFloat(req.query.lon).toFixed(3);
    const cacheKey = `skyradar:meteo:${lat}:${lon}`;

    const cached = await redis.get(cacheKey);
    if (cached) return res.json({ source: 'redis-cache', data: JSON.parse(cached) });

    const url = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}` +
      `&hourly=temperature_2m,relative_humidity_2m,dew_point_2m,cloud_cover,cloud_cover_low,cloud_cover_mid,cloud_cover_high,visibility,wind_speed_10m,precipitation_probability` +
      `&timezone=Asia%2FBangkok`;

    const response = await fetch(url);
    const data = await response.json();
    await redis.setex(cacheKey, 900, JSON.stringify(data)); // Cache 15 phút
    res.json({ source: 'open-meteo-live', data });
  } catch (err) {
    res.status(500).json({ error: 'Không thể lấy dữ liệu khí tượng', details: err.message });
  }
});

// 3. GET /api/v1/spots - Lấy danh sách điểm săn ảnh kèm khoảng cách PostGIS
app.get('/api/v1/spots', async (req, res) => {
  const { region } = req.query;
  const query = region
    ? `SELECT id, name, region, province, ST_Y(geom::geometry) AS lat, ST_X(geom::geometry) AS lon, elevation, bortle, description FROM astro_spots WHERE region = $1 ORDER BY bortle ASC`
    : `SELECT id, name, region, province, ST_Y(geom::geometry) AS lat, ST_X(geom::geometry) AS lon, elevation, bortle, description FROM astro_spots ORDER BY bortle ASC`;
  const result = await pgPool.query(query, region ? [region] : []);
  res.json({ spots: result.rows });
});

app.listen(process.env.PORT || 4000, () => console.log('SkyRadar VN API listening on port 4000 with CARTO_API_KEY'));

Cẩm Nang Hướng Dẫn Sử Dụng StarWatch

Cách đọc nhanh các chỉ số thiên văn, bản đồ phương vị, phân tầng mây và thông số chụp thực địa

1. Ba Chỉ Số Săn Ảnh (Thang 0 – 100)

Thanh địa điểm cố định phía trên luôn hiển thị 3 điểm số tính toán theo thời gian thực:

  • 🌌 Ngân Hà & Trời Tối: Đạt 75–100 điểm khi mây tổng < 25%, Mặt Trăng đã lặn hoặc độ sáng trăng thấp, cấp Bortle ≤ 4 và Tâm Ngân Hà cao > 15°.
  • 🌅 Ráng Bình Minh · Hoàng Hôn: Đạt điểm cao khi Mây Cao/Trung đạt 30%–70% (đóng vai trò màn hứng ánh sáng đỏ vàng) và Mây Thấp < 35% (không che khuất đường chân trời).
  • ☁️ Săn Biển Mây: Tính toán cao độ đáy mây ngưng tụ LCL = 125 × (T - T_dew). Khi độ cao đỉnh núi vượt lên trên đáy mây thấp, độ ẩm > 85% và gió nhẹ < 12km/h, biển mây sẽ đặc và ổn định.
2. Cách Đọc Vòm Thiên Đỉnh 360° & Tia Phương Vị
  • Tâm vòng tròn (90°): Đỉnh đầu người quan sát (Thiên đỉnh - Zenith).
  • Vòng tròn ngoài cùng (0°): Đường chân trời theo 8 hướng địa bàn (B, ĐB, Đ, ĐN, N, TN, T, TB).
  • Tia Phương Vị trên Bản đồ: 3 đường nét đứt phóng ra từ điểm ghim cho biết chính xác hướng máy ảnh cần xoay tới để đón Mặt Trời mọc/lặn (vàng), Mặt Trăng (xanh) và Thiên thể sâu (tím).
  • Biểu đồ Độ Cao 24h: Nhấp trực tiếp vào bất kỳ vị trí nào trên đồ thị hoặc kéo thanh trượt giờ để tua nhanh quỹ đạo bầu trời.
3. Ma Trận Mây 3 Tầng, Cấp Bortle & Điểm Sương
  • Thang Bortle (1 – 9): Đo mức độ ô nhiễm ánh sáng. Bortle 1–3 là bầu trời tối nguyên sơ thấy rõ cấu trúc Ngân Hà bằng mắt thường; Bortle 4–5 là vùng ven/núi gần thị trấn; Bortle 7–9 là nội đô sáng rực.
  • Phân tầng mây: Nếu đứng trên đỉnh núi cao 2.000m (như Tà Xùa, Fansipan), Mây Thấp (<2km) đạt 80% nhưng Mây Trung & Cao bằng 0% nghĩa là bạn đang đứng trên biển mây dưới bầu trời sao trong vắt!
  • 💧 Cảnh báo Đọng Sương (Dew Point): Khi chênh lệch Nhiệt độ - Điểm sương ≤ 2.5°C, hơi nước sẽ ngưng tụ làm mờ thấu kính máy ảnh. Hãy quấn dây sưởi lens (Dew Heater).
4. Tìm Kiếm Tự Do & Hướng Dẫn Chụp Đa Thiết Bị
  • Tìm điểm săn ảnh toàn cầu: Gõ bất kỳ tên núi, đèo, hồ, xã/huyện nào vào ô "Tìm điểm săn ảnh...", hệ thống sẽ truy vấn bản đồ trực tuyến để lấy tọa độ, độ cao và dự báo ngay lập tức.
  • Setup Chụp Thiên Văn: Cuộn xuống mục Hướng Dẫn Setup Chụp Thiên Văn và chọn giữa Máy ảnh Full-Frame, iPhone 18 Pro hoặc Android Flagship để xem thông số ống kính (1x/0.5x/Tele, mm), ISO, tốc độ màn trập phù hợp nhất.
💡 Mẹo: Bạn có thể bật chế độ Đèn Đỏ (Night Vision) trong phần Cài đặt khi đang chụp đêm ngoài thực địa.
